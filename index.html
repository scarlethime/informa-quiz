    let currentQuestion = 0;

    let userAnswers = [];


    function startQuiz() {

        document.getElementById('start-screen').classList.add('hidden');

        document.getElementById('quiz-screen').classList.remove('hidden');

        showQuestion();

    }


    function showQuestion() {

        const q = questions[currentQuestion];

        document.getElementById('question-number').innerText = `ข้อ ${currentQuestion + 1} / 10`;

        document.getElementById('question-text').innerText = q.text;

        

        const optionsContainer = document.getElementById('options-container');

        optionsContainer.innerHTML = ''; 

        

        q.options.forEach(option => {

            const btn = document.createElement('button');

            btn.className = 'btn';

            btn.innerText = option.text;

            btn.onclick = () => selectOption(option.value);

            optionsContainer.appendChild(btn);

        });

    }


    function selectOption(value) {

        userAnswers.push(value); 

        currentQuestion++;

        

        if (currentQuestion < questions.length) {

            showQuestion();

        } else {

            calculateResult();

        }

    }


    function calculateResult() {

        // 1. นับคะแนน

        let counts = { A: 0, B: 0, C: 0 };

        userAnswers.forEach(answer => counts[answer]++);


        // 2. หาตัวเลือกที่มากที่สุด

        let maxCount = Math.max(counts.A, counts.B, counts.C);

        let topChoices = Object.keys(counts).filter(k => counts[k] === maxCount);

        

        let dominantChoice = '';


        // 3. กรณีคะแนนสูสี ให้ใช้ข้อ 10 ตัดสิน

        if (topChoices.length > 1) {

            let q10Answer = userAnswers[9]; // ข้อ 10 คือ index 9

            if (topChoices.includes(q10Answer)) {

                dominantChoice = q10Answer;

            } else {

                dominantChoice = topChoices[0]; // สำรองกรณีข้อ 10 ไม่ได้อยู่ใน topChoices

            }

        } else {

            dominantChoice = topChoices[0];

        }


        // 4. แยกสายตามเงื่อนไข

        let finalCharacter = '';


        if (dominantChoice === 'A') {

            // สลับผลลัพธ์ระหว่าง Sales และ Event (ใช้ข้อ 2 และ 4 เป็นตัวแยก)

            if (userAnswers[1] === 'A' && userAnswers[3] === 'A') { // ข้อ 2 คือ index 1, ข้อ 4 คือ index 3

                finalCharacter = 'sales';

            } else {

                finalCharacter = 'event';

            }

        } else if (dominantChoice === 'B') {

            // สลับผลลัพธ์ระหว่าง Operations และ Support (ใช้ข้อ 3 และ 8 เป็นตัวแยก)

            if (userAnswers[2] === 'B' && userAnswers[7] === 'B') { // ข้อ 3 คือ index 2, ข้อ 8 คือ index 7

                finalCharacter = 'operations';

            } else {

                finalCharacter = 'support';

            }

        } else if (dominantChoice === 'C') {

            finalCharacter = 'marketing';

        }


        showResultScreen(finalCharacter);

    }


    function showResultScreen(characterKey) {

        document.getElementById('quiz-screen').classList.add('hidden');

        document.getElementById('result-screen').classList.remove('hidden');

        

        const result = characters[characterKey];

        

        document.getElementById('result-image').src = result.img;

        document.getElementById('result-career').innerText = result.title;

        document.getElementById('result-dept').innerText = result.dept;

        document.getElementById('result-desc').innerHTML = result.desc;

    }
